cmake_minimum_required(VERSION 3.20)
project(manta_lcd4214 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(UNICORN_ROOT "" CACHE PATH "Directory containing Unicorn include/ and lib/")
find_path(UNICORN_INCLUDE unicorn/unicorn.h HINTS "${UNICORN_ROOT}/include" REQUIRED)
find_library(UNICORN_LIBRARY NAMES unicorn unicorn.lib HINTS "${UNICORN_ROOT}/lib" REQUIRED)
add_library(manta_core src/cpu/engine.cpp src/board/machine.cpp src/peripherals/riu.cpp src/storage/flash.cpp src/storage/sha256.cpp)
target_include_directories(manta_core PUBLIC src "${UNICORN_INCLUDE}")
target_link_libraries(manta_core PUBLIC "${UNICORN_LIBRARY}")
if(MSVC)
 target_compile_options(manta_core PRIVATE /W4)
else()
 target_compile_options(manta_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(manta src/main.cpp src/ui/native_window.cpp)
target_link_libraries(manta PRIVATE manta_core)
if(WIN32)
 target_link_libraries(manta PRIVATE user32 gdi32 comdlg32 shell32)
 if(MINGW)
  target_link_options(manta PRIVATE -static-libgcc -static-libstdc++)
 endif()
 add_custom_command(TARGET manta POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${UNICORN_ROOT}/lib/unicorn.dll" $<TARGET_FILE_DIR:manta>)
endif()
include(CTest)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE manta_core)
if(MINGW)
 target_link_options(core_tests PRIVATE -static-libgcc -static-libstdc++)
endif()
add_test(NAME core_semantics COMMAND core_tests)
