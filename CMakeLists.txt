cmake_minimum_required(VERSION 3.20)
project(manta_lcd4214 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(UNICORN_ROOT "" CACHE PATH "Directory containing Unicorn include/ and lib/")
find_path(UNICORN_INCLUDE unicorn/unicorn.h HINTS "${UNICORN_ROOT}/include" REQUIRED)
find_library(UNICORN_LIBRARY NAMES unicorn unicorn.lib HINTS "${UNICORN_ROOT}/lib" REQUIRED)
add_library(manta_core src/cpu/engine.cpp src/cpu/openrisc.cpp src/board/machine.cpp src/peripherals/riu.cpp src/peripherals/gpio_i2c.cpp src/storage/flash.cpp src/storage/sha256.cpp src/snapshot/snapshot.cpp src/profile/sampler.cpp)
set(EMU8051_SOURCES src/cpu/third_party/emu8051/core.c src/cpu/third_party/emu8051/opcodes.c src/cpu/third_party/emu8051/disasm.c)
set_source_files_properties(${EMU8051_SOURCES} PROPERTIES LANGUAGE CXX)
add_library(manta_emu8051 STATIC ${EMU8051_SOURCES})
target_sources(manta_core PRIVATE src/cpu/vd_mcu.cpp)
target_link_libraries(manta_core PUBLIC manta_emu8051)
target_include_directories(manta_core PUBLIC src "${UNICORN_INCLUDE}")
target_link_libraries(manta_core PUBLIC "${UNICORN_LIBRARY}")
if(MSVC)
 target_compile_options(manta_core PRIVATE /W4)
else()
 target_compile_options(manta_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(manta src/main.cpp src/ui/native_window.cpp src/ui/frontend_host.cpp)
target_link_libraries(manta PRIVATE manta_core)
if(WIN32)
 target_link_libraries(manta PRIVATE user32 gdi32 comdlg32 shell32)
 if(MINGW)
  target_link_options(manta PRIVATE -static-libgcc -static-libstdc++)
 endif()
 # Wheels call the DLL unicorn.dll; MinGW source builds call it libunicorn.dll.
 file(GLOB UNICORN_RUNTIME_DLLS "${UNICORN_ROOT}/lib/*.dll")
 foreach(runtime IN LISTS UNICORN_RUNTIME_DLLS)
  add_custom_command(TARGET manta POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${runtime}" $<TARGET_FILE_DIR:manta>)
 endforeach()

endif()
# Native port (side project): the same machine with the statically recompiled firmware as its CPU. Needs native/gen (python native/tools/mips_recomp.py).
if(EXISTS "${CMAKE_SOURCE_DIR}/native/gen/tables.cpp")
 file(GLOB MANTA_NATIVE_GEN "${CMAKE_SOURCE_DIR}/native/gen/*.cpp")
 set_source_files_properties(${MANTA_NATIVE_GEN} PROPERTIES COMPILE_OPTIONS "-O1;-fno-strict-aliasing;-w")
 add_executable(manta_native_full src/main.cpp src/ui/native_window.cpp src/ui/frontend_host.cpp native/runtime/runtime.cpp native/runtime/native_runner.cpp native/runtime/pages.cpp native/runtime/page_osk.cpp native/runtime/page_net.cpp native/runtime/page_browser.cpp native/runtime/http.cpp native/runtime/html_text.cpp ${MANTA_NATIVE_GEN})
 target_include_directories(manta_native_full PRIVATE native/runtime)
 target_compile_definitions(manta_native_full PRIVATE MANTA_NATIVE_RUNNER)
 target_link_libraries(manta_native_full PRIVATE manta_core)
 if(WIN32)
  target_link_libraries(manta_native_full PRIVATE user32 gdi32 comdlg32 shell32)
  if(MINGW)
   target_link_options(manta_native_full PRIVATE -static-libgcc -static-libstdc++)
  endif()
  foreach(runtime IN LISTS UNICORN_RUNTIME_DLLS)
   add_custom_command(TARGET manta_native_full POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${runtime}" $<TARGET_FILE_DIR:manta_native_full>)
  endforeach()
 endif()
endif()
